import 'package:flutter/material.dart';

class Tutor {
  final String name;
  final String imageUrl;
  final String hoursAvailable;

  Tutor({required this.name, required this.imageUrl, required this.hoursAvailable});
}

class TutorSelectionModal extends StatelessWidget {
  final List<Tutor> tutors;

  TutorSelectionModal({required this.tutors});

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.all(16.0),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text(
            'Select a tutor to schedule',
            style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold),
          ),
          SizedBox(height: 10),
          ListView.separated(
            shrinkWrap: true,
            itemCount: tutors.length,
            separatorBuilder: (context, index) => Divider(),
            itemBuilder: (context, index) {
              return ListTile(
                leading: CircleAvatar(
                  backgroundImage: AssetImage(tutors[index].imageUrl),
                ),
                title: Text(tutors[index].name),
                subtitle: Text(tutors[index].hoursAvailable),
                trailing: Icon(Icons.arrow_forward_ios),
                onTap: () {
                  // Handle tutor selection
                  Navigator.pop(context);
                },
              );
            },
          ),
        ],
      ),
    );
  }
}




import 'package:flutter/material.dart';
import 'tutor_selection_modal.dart'; // Import the file containing the TutorSelectionModal class

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {
  final List<Tutor> tutors = [
    Tutor(name: 'Safwan', imageUrl: 'assets/images/safwan.jpg', hoursAvailable: '0 hours to schedule'),
    Tutor(name: 'Wade Warren', imageUrl: 'assets/images/wade_warren.jpg', hoursAvailable: '0 hours to schedule'),
    Tutor(name: 'Cameron Williamson', imageUrl: 'assets/images/cameron_williamson.jpg', hoursAvailable: '0 hours to schedule'),
  ];

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(title: Text('Tutor Selection')),
        body: Center(
          child: ElevatedButton(
            onPressed: () => showModalBottomSheet(
              context: context,
              shape: RoundedRectangleBorder(
                borderRadius: BorderRadius.vertical(top: Radius.circular(20)),
              ),
              builder: (context) => TutorSelectionModal(tutors: tutors),
            ),
            child: Text('Schedule Tutor'),
          ),
        ),
      ),
    );
  }
}
